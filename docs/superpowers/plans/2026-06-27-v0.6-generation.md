# IronLog-V2 v0.6 — Generation Loop Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the between-session generation loop — the deterministic spine that resolves context, lays a skeleton, takes selections from a swappable proposer, assembles a full `Session`, validates/repairs/falls-back, and commits at approval — with the live Gemini proposer wired last.

**Architecture:** A new `ironlog/generation/` subpackage orchestrates the **already-built** engine (`validator`, `ledger`, `loading`, `progression`, `calibration`, `run_analysis`, `stall`). The LLM is confined to a single `Proposer.propose(payload) -> Selections` call behind a port; Tasks 1–10 build and test the entire deterministic spine against a `StubProposer` returning canned selections; Task 11 adds the real `gemini-3.1-flash-lite` HTTP adapter. The model emits selections only (movement ids + ordering + variant + technique tags + rationale); the engine computes every number; `current_load` is written only at approval.

**Tech Stack:** Python 3 / SQLModel / FastAPI / pytest. New runtime dep (Task 11 only): `httpx`. Source of truth: `docs/superpowers/specs/2026-06-27-v0.6-generation-design.md` (Forks 1–7 + refinement tier).

- **BUILD-AND-TEST-ONLY:** never run `python -m ironlog.seed`; never write prod `~/projects/IronLog-V2/ironlog.db`. Tests seed an in-memory DB.
- **Tests run on myflix only:** `ssh myflix 'cd ~/projects/IronLog-V2 && .venv/bin/pytest -q'` (workstation venv lacks deps). Baseline before this plan: **149 passing**.
- **No `from __future__ import annotations`** in any file.
- **Architecture invariants** (repo `CLAUDE.md`): `engine/` stays pure (the new `generation/` package is the orchestration layer, may touch the DB); definition-vs-state; planned-vs-logged; mandatory `feedback_tap`; rules-dispose/model-proposes; **two-writer boundary** — `run_analysis` writes outcome fields (`e1rm`/`e1rmhistory`/`calibration_status`/counters), generation writes only `current_load`, and only at approve.
- **Selections-only contract (Fork 1):** the proposer output schema has NO numeric prescription fields. Enforced by the `SELECTIONS_JSON_SCHEMA` (Task 1) and the Gemini `response_json_schema` (Task 11).
- **Migration authoring rule:** any `deploy/migrations/NNN_*.sql` must be single-statement-atomic OR fully idempotent (`IF NOT EXISTS`).
- **Reuse, don't reinvent (Fork 6 / Principle 5):** the assembler and fallback call existing `loading`/`progression`/`calibration` functions; fresh-movement starting load routes to the calibration resolver — no duplicate load logic.

"""proposer.py — the selections-only contract (Fork 1) + the proposer port (Fork 2)."""
from dataclasses import dataclass, field
from typing import List, Optional, Protocol, runtime_checkable

# The §6 response_json_schema. Gemini structured output is constrained to this
# shape (Task 11), so "no numeric prescription fields" holds at the API boundary.
SELECTIONS_JSON_SCHEMA = {
    "type": "object",
    "properties": {
        "ordering": {"type": "array", "items": {"type": "string"}},
        "slots": {
            "type": "array",
            "items": {
                "type": "object",
                "properties": {
                    "slot_id": {"type": "string"},
                    "movement_id": {"type": "integer"},
                    "variant": {"type": "string", "nullable": True},
                    "technique_tags": {"type": "array", "items": {"type": "string"}},
                },
                "required": ["slot_id", "movement_id"],
            },
        },
        "rationale": {"type": "string"},
    },
    "required": ["ordering", "slots", "rationale"],
}

@dataclass
class SlotSelection:
    slot_id: str
    movement_id: int
    variant: Optional[str] = None
    technique_tags: List[str] = field(default_factory=list)

@dataclass
class Selections:
    ordering: List[str]
    slots: List[SlotSelection]
    rationale: str

def selections_from_dict(d: dict) -> Selections:
    return Selections(
        ordering=list(d.get("ordering", [])),
        slots=[SlotSelection(
            slot_id=s["slot_id"], movement_id=int(s["movement_id"]),
            variant=s.get("variant"), technique_tags=list(s.get("technique_tags", [])),
        ) for s in d.get("slots", [])],
        rationale=d.get("rationale", ""),
    )

@runtime_checkable
class Proposer(Protocol):
    def propose(self, payload: dict) -> Selections: ...

class StubProposer:
    """Deterministic test/offline proposer: returns canned selections verbatim."""
    def __init__(self, canned: Selections):
        self._canned = canned
    def propose(self, payload: dict) -> Selections:
        return self._canned
```

"""skeleton.py — lay the fixed deterministic skeleton (docs/06 §3 step 2)."""
from dataclasses import dataclass, field
from typing import List

from sqlmodel import Session, select

from ..models.enums import Status
from ..models.library import Movement
from .daytemplate import DAY_TEMPLATES, SlotSpec

@dataclass
class Skeleton:
    day_role: str
    anchor_movement_ids: List[int] = field(default_factory=list)
    adaptive_slots: List[SlotSpec] = field(default_factory=list)

def lay_skeleton(day_role: str, db: Session) -> Skeleton:
    tpl = DAY_TEMPLATES[day_role]
    anchors: List[int] = []
    for base in tpl.primary_base_names:
        m = db.exec(
            select(Movement).where(
                Movement.base_name == base,
                Movement.is_primary == True,        # noqa: E712
                Movement.status == Status.ACTIVE,
            )
        ).first()
        if m is None:
            raise ValueError(f"no ACTIVE primary movement for base_name {base!r}")
        anchors.append(m.id)
    return Skeleton(day_role=day_role, anchor_movement_ids=anchors,
                    adaptive_slots=list(tpl.slots))
```

"""context.py — RESOLVE CONTEXT (Fork 2 inject) + candidate menus (Fork 3)."""
from dataclasses import dataclass, field
from datetime import date
from typing import Callable, Dict, List, Optional, Set

from sqlmodel import Session, select

from ..engine.ledger import compute_tallies
from ..engine.stall import detect_stall
from ..engine.validator import WeeklyTallies
from ..models.enums import Objective, Status
from ..models.library import (
    E1rmHistory, EngineState, Movement, MovementState, PhasePolicy,
)
from ..models.session import SetLog
from ..persistence.run_analysis import select_progress_window
from .daytemplate import SlotSpec
from .skeleton import Skeleton

# Coarse pattern -> lift_category map (mirrors ledger's pull/push stand-in;
# v0.6 keeps it simple, post-beta taxonomy may supersede). Patterns the
# templates use map to the library's LiftCategory / knee_modality columns.
_PATTERN_LIFT_CATEGORIES: Dict[str, Set[str]] = {
    "horizontal_pull": {"ROW"},
    "vertical_push": {"OHP", "CG_PRESS"},
    "hip_hinge": {"RDL", "DEADLIFT", "HIP_THRUST"},
    "knee_flexion": {"NONE"},  # accessory hams/quads — broad; refine post-beta
}

# Weekly knee-frequency targets (docs/06 §4).
KNEE_TARGETS = {"NORDIC": 2, "TIB": 2, "KOT": 2, "SISSY": 1}
PULL_PUSH_TARGET = 2.0

@dataclass
class GenerationContext:
    phase: str
    phase_policy: PhasePolicy
    manifest_equipment_ids: Set[int]
    movement_states: Dict[int, MovementState]
    tallies: WeeklyTallies
    owed: dict
    recent_signatures: List[dict]
    weak_point_hints: Dict[int, str]
    candidate_menus: Dict[str, List[int]] = field(default_factory=dict)

def build_candidate_menu(slot: SlotSpec, db: Session,
                         manifest_ids: Set[int]) -> List[int]:
    """HARD filter (Fork 3): ACTIVE + equipment-feasible + tier/pattern.
    The model cannot pick off-menu; menu-membership is re-checked by validate."""
    rows = db.exec(select(Movement).where(Movement.status == Status.ACTIVE)).all()
    out: List[int] = []
    for m in rows:
        if m.is_primary:
            continue  # anchors are placed deterministically, never in adaptive menus
        if m.load_equipment_id is not None and m.load_equipment_id not in manifest_ids:
            continue
        if slot.kind == "knee":
            if m.knee_modality is None or m.knee_modality.value != slot.knee_modality:
                continue
        elif slot.pattern is not None:
            cats = _PATTERN_LIFT_CATEGORIES.get(slot.pattern, set())
            if m.lift_category.value not in cats:
                continue
        out.append(m.id)
    return sorted(out)

def compute_owed_requirements(tallies: WeeklyTallies) -> dict:
    owed_knee = {k: max(0, tallies.knee_targets.get(k, 0) - tallies.knee_counts.get(k, 0))
                 for k in tallies.knee_targets}
    push = tallies.push_volume or 0.0
    ratio = (tallies.pull_volume / push) if push else None
    return {"knee_owed": owed_knee,
            "pull_push_ratio": ratio,
            "pull_push_target": tallies.pull_push_target}

def build_weak_point_hints(db: Session) -> Dict[int, str]:
    """L1 soft hint (Fork 3 / §9): for each movement with PROGRESS history, run
    detect_stall over its PROGRESS window; if stalled, flag the limiter (soft)."""
    hints: Dict[int, str] = {}
    states = db.exec(select(MovementState)).all()
    for st in states:
        rows = db.exec(
            select(E1rmHistory).where(E1rmHistory.movement_id == st.movement_id)
        ).all()
        window = select_progress_window(list(rows))
        if not window:
            continue
        sig = detect_stall(window, st.consecutive_failed_progressions, Objective.PROGRESS)
        if sig.stalled:
            hints[st.movement_id] = "stalled: bias accessory volume toward the limiter (L1)"
    return hints

def resolve_context(day_role: str, skeleton: Skeleton, db: Session,
                    week_keyer: Callable[[date], object]) -> GenerationContext:
    engine_state = db.exec(select(EngineState)).one()
    phase = engine_state.current_phase
    policy = db.exec(select(PhasePolicy).where(PhasePolicy.phase == phase)).one()
    movements = {m.id: m for m in db.exec(select(Movement)).all()}
    manifest_ids = {m.load_equipment_id for m in movements.values()
                    if m.load_equipment_id is not None}
    states = {s.movement_id: s for s in db.exec(select(MovementState)).all()}

    set_logs = db.exec(select(SetLog)).all()
    tallies = compute_tallies(set_logs, movements)
    tallies.knee_targets = dict(KNEE_TARGETS)
    tallies.pull_push_target = PULL_PUSH_TARGET

    menus = {}
    for slot in skeleton.adaptive_slots:
        if slot.kind in ("giant", "knee"):
            menus[slot.slot_id] = build_candidate_menu(slot, db, manifest_ids)

    recent = [s.signature for s in _recent_same_role_sessions(db, day_role)]
    return GenerationContext(
        phase=phase, phase_policy=policy, manifest_equipment_ids=manifest_ids,
        movement_states=states, tallies=tallies,
        owed=compute_owed_requirements(tallies),

"""signature.py — session novelty signature (§7, simple beta form)."""
from typing import Dict, List

_WEIGHTS = {"exercise_set": 0.40, "rep_zone": 0.25, "techniques": 0.20,
            "pattern": 0.10, "order": 0.05}

def compute_signature(slot_movement_ids: List[int], rep_zones: List[str],
                      technique_tags: List[str], patterns: List[str],
                      ordering: List[str]) -> dict:
    return {"exercise_set": sorted(set(slot_movement_ids)),
            "rep_zone": sorted(set(rep_zones)),
            "techniques": sorted(set(technique_tags)),
            "pattern": sorted(set(patterns)),
            "order": list(ordering)}

def _jaccard_distance(a: list, b: list) -> float:
    sa, sb = set(a), set(b)
    if not sa and not sb:
        return 0.0
    return 1.0 - len(sa & sb) / len(sa | sb)

def _order_distance(a: list, b: list) -> float:
    if not a and not b:
        return 0.0
    n = max(len(a), len(b))
    same = sum(1 for i in range(min(len(a), len(b))) if a[i] == b[i])
    return 1.0 - same / n

def signature_distance(a: dict, b: dict) -> float:
    d = 0.0
    for axis, w in _WEIGHTS.items():
        if axis == "order":
            d += w * _order_distance(a.get(axis, []), b.get(axis, []))
        else:
            d += w * _jaccard_distance(a.get(axis, []), b.get(axis, []))
    return d

def meets_novelty(candidate: dict, recents: List[dict],
                  threshold: float = 0.30) -> bool:
    """Soft (§7): differ >= threshold from EACH recent same-day_role signature."""
    return all(signature_distance(candidate, r) >= threshold for r in recents)
```

from typing import Dict

@dataclass

    session: Session
    prospective_current_loads: Dict[int, float] = field(default_factory=dict)

def resolve_start_load(movement: Movement, state: MovementState, db: DBSession) -> float:
    """Single source for a movement's working load (Pin 1 — routes to existing
    fields). If state.current_load is set, use it; else a fresh movement starts
    from start_ratio*anchor_e1rm (ratio-variant) or its load_floor."""
    if state is not None and state.current_load is not None:
        return state.current_load
    if movement.start_ratio is not None and movement.derived_from_id is not None:
        anchor_state = db.exec(
            select(MovementState).where(
                MovementState.movement_id == movement.derived_from_id)
        ).first()
        if anchor_state is not None and anchor_state.e1rm is not None:
            return movement.start_ratio * anchor_state.e1rm
    return movement.load_floor if movement.load_floor is not None else 0.0

def _step_and_floor(movement: Movement, db: DBSession):
    eq = db.get(Equipment, movement.load_equipment_id) if movement.load_equipment_id else None
    step = movement.min_step or (eq.min_step if eq else 2.5) or 2.5
    floor = movement.load_floor if movement.load_floor is not None else (eq.load_floor if eq else None)
    return step, floor

def assemble(selections: Selections, skeleton: Skeleton,
             ctx: GenerationContext, db: DBSession) -> AssembledSession:
    from datetime import date
    movements = {m.id: m for m in db.exec(select(Movement)).all()}
    session = Session(date=date.today(), day_role=skeleton.day_role,
                      phase=ctx.phase, status=SessionStatus.PLANNED,
                      rationale=selections.rationale)
    prospective: Dict[int, float] = {}
    order = 0

    # 1) anchors first as STRAIGHT T1 groups
    for mid in skeleton.anchor_movement_ids:
        _add_group(session, db, ctx, movements[mid], GroupType.STRAIGHT,
                   order, prospective, is_anchor=True)
        order += 1
    # 2) the adaptive layer in the model's chosen ordering
    selected = {s.slot_id: s for s in selections.slots}
    giant = ExerciseGroup(order_index=order, group_type=GroupType.GIANT_SET, rounds=3)
    has_giant = False
    for slot_id in selections.ordering:
        sel = selected.get(slot_id)
        if sel is None:
            continue
        slot = next((s for s in skeleton.adaptive_slots if s.slot_id == slot_id), None)
        if slot is None:
            continue
        m = movements[sel.movement_id]
        if slot.kind == "giant":
            _add_exercise(giant, db, ctx, m, len(giant.exercises), prospective)
            has_giant = True
        else:  # knee / conditioning -> own STRAIGHT group
            _add_group(session, db, ctx, m, GroupType.STRAIGHT, order, prospective)
            order += 1
    if has_giant:
        session.groups.append(giant)
    return AssembledSession(session=session, prospective_current_loads=prospective)

def _add_group(session, db, ctx, movement, group_type, order, prospective, is_anchor=False):
    g = ExerciseGroup(order_index=order, group_type=group_type,
                      rounds=1 if group_type == GroupType.STRAIGHT else 3)
    _add_exercise(g, db, ctx, movement, 0, prospective, is_anchor=is_anchor)
    session.groups.append(g)

def _add_exercise(group, db, ctx, movement, ex_order, prospective, is_anchor=False):
    state = ctx.movement_states.get(movement.id)
    objective = resolve_objective(movement.objective_override,
                                  ctx.phase_policy.default_objective)
    ex = PlannedExercise(movement_id=movement.id, order_index=ex_order,
                         scheme=movement.scheme, objective=objective)
    step, floor = _step_and_floor(movement, db)
    base = resolve_start_load(movement, state, db)
    load = clamp_to_cap(round_to_achievable(base, floor, step), movement.cap)
    prospective[movement.id] = load
    sets = _sets_for_scheme(movement.scheme, load, ctx, is_anchor)
    ex.planned_sets.extend(sets)
    group.exercises.append(ex)

def _sets_for_scheme(scheme, load, ctx, is_anchor):
    pol = ctx.phase_policy
    if scheme == Scheme.TOPSET_BACKOFF:
        return [
            PlannedSet(set_index=0, set_role=SetRole.TOP, target_load=load,
                       target_reps_low=3, target_reps_high=5, target_rpe=pol.top_set_rpe),
            PlannedSet(set_index=1, set_role=SetRole.BACKOFF,
                       target_load=round(load * 0.9, 1),
                       target_reps_low=5, target_reps_high=8, target_rpe=pol.rpe_band_low),
        ]
    return [PlannedSet(set_index=i, set_role=SetRole.WORKING, target_load=load,
                       target_reps_low=8, target_reps_high=12,
                       target_rpe=pol.rpe_band_high) for i in range(3)]

"""repair.py — validate + bounded repair (Fork 4 a/b)."""
from dataclasses import dataclass, field
from typing import Callable, Dict, List, Optional

from sqlmodel import Session as DBSession, select

from ..engine.validator import (
    MovementInfo, RuleCode, ValidationContext, ValidationResult, Violation,
    ViolationKind, validate,
)
from ..models.library import Movement
from .assembler import AssembledSession, assemble

def build_validation_context(ctx: GenerationContext, db: DBSession) -> ValidationContext:
    infos: Dict[int, MovementInfo] = {}
    for m in db.exec(select(Movement)).all():
        infos[m.id] = MovementInfo(
            movement_id=m.id, is_primary=m.is_primary,
            load_equipment_id=m.load_equipment_id, load_floor=m.load_floor,
            cap=m.cap, rpe_cap_exempt=m.rpe_cap_exempt,
            lift_category=m.lift_category, progression_mode=m.progression_mode,
        )
    vc = ValidationContext(
        movements=infos, manifest_equipment_ids=set(ctx.manifest_equipment_ids),
        phase_hard_cap=ctx.phase_policy.hard_cap, tallies=ctx.tallies,
    )
    return vc

def rejection_reasons(result: ValidationResult) -> List[str]:
    """Outcome-only (Fork 4a): the validator's message states requirement+status
    +locus. We pass it through verbatim and NEVER append a remedy."""
    out = []
    for v in result.rejects:
        locus = ""
        if v.group_index is not None:
            locus = f" [group {v.group_index}" + (
                f", movement {v.movement_id}]" if v.movement_id is not None else "]")
        out.append(f"{v.rule.value}: {v.message}{locus}")
    return out

def apply_clamps(session, result: ValidationResult) -> int:
    """Apply each CLAMP's corrected_value via the validator's (locator, rule)
    contract: LOAD_* -> target_load, RPE_OVER_CAP -> target_rpe."""
    index = {}
    for g in session.groups:
        for e in g.exercises:
            for ps in e.planned_sets:
                index[(g.order_index, e.movement_id, ps.set_index)] = ps
    n = 0
    for c in result.clamps:
        ps = index.get((c.group_index, c.movement_id, c.set_index))
        if ps is None or c.corrected_value is None:
            continue
        if c.rule in (RuleCode.LOAD_BELOW_FLOOR, RuleCode.LOAD_OVER_CAP):
            ps.target_load = c.corrected_value
        elif c.rule == RuleCode.RPE_OVER_CAP:
            ps.target_rpe = c.corrected_value
        n += 1
    return n

@dataclass
class RepairOutcome:
    assembled: Optional[AssembledSession]
    attempts: int
    clamps_applied: int
    rejections: List[str] = field(default_factory=list)
    exhausted: bool = False

def propose_validate_repair(proposer: Proposer, payload: dict, skeleton: Skeleton,
                            ctx: GenerationContext, db: DBSession,
                            max_retries: int = 3) -> RepairOutcome:
    vc = build_validation_context(ctx, db)
    reasons: List[str] = []
    total_clamps = 0
    for attempt in range(1, max_retries + 1):
        payload_with_feedback = dict(payload)
        if reasons:
            payload_with_feedback["rejections"] = reasons  # outcome-only feedback
        sel = proposer.propose(payload_with_feedback)
        assembled = assemble(sel, skeleton, ctx, db)
        result = validate(assembled.session, vc)
        total_clamps += apply_clamps(assembled.session, result)
        result = validate(assembled.session, vc)  # re-validate after clamps
        if result.is_structurally_valid:
            return RepairOutcome(assembled, attempt, total_clamps, [], False)
        reasons = rejection_reasons(result)
    return RepairOutcome(None, max_retries, total_clamps, reasons, True)
```

def cold_start_selections(skeleton: Skeleton, ctx: GenerationContext) -> Selections:
    """Deterministic, trainable: fill each adaptive slot with the first candidate

    slots = []

def last_valid_selections(skeleton: Skeleton, ctx: GenerationContext,
                          db: DBSession) -> Optional[Selections]:
    prior = db.exec(
        select(WorkoutSession)
        .where(WorkoutSession.day_role == skeleton.day_role,
               WorkoutSession.status == SessionStatus.COMPLETED)
        .order_by(WorkoutSession.date.desc())

           for e in sorted(g.exercises, key=lambda e: e.order_index)]
    non_anchor = exs[-len(slot_iter):] if len(exs) >= len(slot_iter) else exs
    for spec, ex in zip(slot_iter, non_anchor):
        slots.append(SlotSelection(slot_id=spec.slot_id, movement_id=ex.movement_id))
        order.append(spec.slot_id)
    if not slots:
        return None
    return Selections(ordering=order, slots=slots,
                      rationale="deterministic fallback (last valid, loads refreshed)")

def fallback_session(skeleton: Skeleton, ctx: GenerationContext,
                     db: DBSession) -> AssembledSession:
    sel = last_valid_selections(skeleton, ctx, db) or cold_start_selections(skeleton, ctx)
    return assemble(sel, skeleton, ctx, db)  # re-assemble => loads refresh from state
```

from datetime import datetime
from typing import List

from sqlmodel import Session as DBSession, select

from ..models.enums import SessionStatus
from ..models.library import GenerationLog, MovementState
from ..models.session import Session
from .assembler import AssembledSession
from .repair import RepairOutcome

def is_clean(outcome: RepairOutcome) -> bool:
    """Fork 7a: clean == zero repairs AND zero clamps. A fallback (outcome is
    None / exhausted) is never clean (7b derived)."""
    if outcome.assembled is None or outcome.exhausted:
        return False
    return outcome.attempts == 1 and outcome.clamps_applied == 0

def commit_session(assembled: AssembledSession, db: DBSession, *, approval_mode: str,
                   prompt: dict, selections_dict: dict, clamps: List, repairs: List,
                   fallback_used: bool) -> Session:
    """The SOLE writer of current_load (Fork 7c). Persists the session graph,
    writes current_load from prospective values, writes provenance."""
    session = assembled.session
    session.status = SessionStatus.PLANNED
    session.approved_at = datetime.utcnow()
    db.add(session)
    db.commit()
    db.refresh(session)
    for mid, load in assembled.prospective_current_loads.items():
        st = db.exec(select(MovementState).where(
            MovementState.movement_id == mid)).first()
        if st is not None:
            st.current_load = load          # the ONLY place generation writes it
            db.add(st)
    db.add(GenerationLog(
        session_id=session.id, prompt_json=prompt, selections_json=selections_dict,
        clamps_json=clamps, repairs_json=repairs, approval_mode=approval_mode,
        fallback_used=fallback_used))
    db.commit()
    db.refresh(session)
    return session
```
